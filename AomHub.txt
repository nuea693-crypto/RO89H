local function saveConfig()

local function getPing()
    local gui = LocalPlayer:FindFirstChild("PlayerGui")
    if not gui then return 0.15 end
    local stats = gui:FindFirstChild("NetworkStats")
    if not stats then return 0.15 end
    local pingLabel = stats:FindFirstChild("PingLabel")
    if not pingLabel then return 0.15 end
    local text = pingLabel.Text
    if type(text) ~= "string" then return 0.15 end
    local num = tonumber(text:match("%d+"))
    if not num then return 0.15 end
    local ping = num / 1000
    if ping < 0 or ping > 1.5 then ping = 0.15 end
    return ping
end

local function getAimTargetPart(char)

    if char:FindFirstChild("Head") then table.insert(parts, char.Head) end
    if char:FindFirstChild("HumanoidRootPart") then table.insert(parts, char.HumanoidRootPart) end

local function getClosestTarget()
    if not F then return nil end
    local closest = nil
    local closestDist = H
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local camPos = Camera.CFrame.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local char = player.Character
            local hum = char:FindFirstChild("Humanoid")
            local aimPart = getAimTargetPart(char)
            if hum and hum.Health > 0 and aimPart then
                local worldPos = aimPart.Position
                local screenPos, onScreen = Camera:WorldToViewportPoint(worldPos)
                if onScreen then
                    local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist2D <= H and not isPlayerExcluded(player.Name) then
                        if not isObstructed(camPos, worldPos, char) then
                            if dist2D < closestDist then
                                closestDist = dist2D
                                closest = player
                            end
                        end
                    end
                end
            end
        end
    end
    return closest
end

local function predictPosition(part, hrp)
    if not part then return Vector3.zero end
    local ping = getPing() * 1.3
    local vel = (hrp and hrp.AssemblyLinearVelocity) or Vector3.zero
    if vel.Y < -100 then
        vel = Vector3.new(vel.X, 0, vel.Z)
    end
    local angularVel = (hrp and hrp.AssemblyAngularVelocity) or Vector3.zero
    if angularVel.Y ~= 0 then
        local rootPos = hrp.Position
        local offset = part.Position - rootPos
        local angle = angularVel.Y * ping
        local cosA, sinA = math.cos(angle), math.sin(angle)
        local newOffset = Vector3.new(offset.X * cosA - offset.Z * sinA, offset.Y, offset.X * sinA + offset.Z * cosA)
        return rootPos + newOffset + (vel * ping)
    end
    return part.Position + (vel * ping)
end

local function NetGet(...)
    if not aC or not aC.func then return end
    local args = {...}
    for i, v in ipairs(args) do
        if typeof(v) == "Instance" and v:IsA("Model") and #v:GetChildren() == 0 then
            local dropped = Workspace:FindFirstChild("DroppedItems")
            if dropped then
                local model = dropped:FindFirstChildWhichIsA("Model")
                if model then
                    args[i] = model
                else
                    return
                end
            else
                return
            end
        end
    end
    aC.func = (aC.func or 0) + 1
    local ok, result = pcall(function()
        if GetRemote then
            return GetRemote:InvokeServer(aC.func, unpack(args))
        end
    end)
    if not ok then warn("[NetGet Error]", result) end
    return result
end

    end

local function getPlayersInRange(range)
    local result = {}
    local myChar = LocalPlayer.Character
    if not myChar or not myChar.PrimaryPart then return result end
    local myPos = myChar.PrimaryPart.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character.PrimaryPart then
            local ok, dist = pcall(function() return (player.Character.PrimaryPart.Position - myPos).Magnitude end)
            if ok and dist and dist <= range then
                table.insert(result, player)
            end
        end
    end
    return result
end

local function AttackNearby()
    if not SendRemote then return end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar.PrimaryPart then return end
    local tool = getActiveTool()
    if not tool or not isMeleeTool(tool) then return end
    if tool.Parent ~= myChar then return end
    local players = getPlayersInRange(ap)
    if #players == 0 then return end
    local myPos = myChar.PrimaryPart.Position
    local targets = {}
    local positions = {}
    for _, player in ipairs(players) do
        if player.Character and player.Character.PrimaryPart then
            local aimPart = getAimTargetPart(player.Character)
            local hrp = player.Character.PrimaryPart
            if aimPart and hrp then
                local pred = predictPosition(aimPart, hrp)
                table.insert(targets, player)
                table.insert(positions, pred)
            end
        end
    end
    if #targets == 0 then return end
    local aimPos = positions[1]
    local cf = CFrame.lookAt(myPos, aimPos)
    local data = {"melee_attack", tool, targets, cf, 0.75}
    pcall(function()
        if aC then
            aC.event = (aC.event or 0) + 1
            SendRemote:FireServer(aC.event, unpack(data))
        else
            aq = (aq or 0) + 1
            SendRemote:FireServer(aq, unpack(data))
        end
    end)
end

local itemCache = {}

function getInventoryItemKey(tool)
    local handle = tool:FindFirstChild("Handle")
    local rarity = tool:GetAttribute("RarityName") or "Common"
    if handle then
        local mesh = handle:FindFirstChildOfClass("SpecialMesh")
        if mesh and mesh.MeshId ~= "" then
            return mesh.MeshId .. (mesh.TextureId or "") .. "_RARITY_" .. rarity
        elseif handle:IsA("MeshPart") and handle.MeshId ~= "" then
            return handle.MeshId .. (handle.TextureID or "") .. "_RARITY_" .. rarity
        end
    end
    local itemId = tool:GetAttribute("ItemId") or tool:GetAttribute("Id") or tool.Name
    if itemId and itemId ~= "" and itemId ~= tool.Name then
        return "ITEMID_" .. itemId .. "_RARITY_" .. rarity
    end
    return "NAME_" .. (tool:GetAttribute("DisplayName") or tool.Name) .. "_" .. tool.Name .. "_RARITY_" .. rarity
end

function getInventoryWeaponInfo(tool)
    if not tool or not tool:IsA("Tool") then return nil end
    return itemCache[getInventoryItemKey(tool)]
end

function registerInventoryItems()
    if not ItemsFolder then return end
    for _, folder in ipairs(ItemsFolder:GetChildren()) do
        if folder:IsA("Folder") then
            for _, tool in ipairs(folder:GetChildren()) do
                if tool:IsA("Tool") then
                    local key = getInventoryItemKey(tool)
                    if key then
                        itemCache[key] = {
                            Name = tool:GetAttribute("DisplayName") or tool.Name,
                            Rarity = tool:GetAttribute("RarityName") or "Common",
                            ToolName = tool.Name
                        }
                    end
                end
            end
        end
    end
end
pcall(registerInventoryItems)

local function setFinishPrompt(prompt)
    if prompt and prompt:IsA("ProximityPrompt") then
        prompt.HoldDuration = ae
        prompt.MaxActivationDistance = 20
    end
end

local function tryHoldPrompt(prompt, duration)
    if not prompt or prompt:GetAttribute("__AutoFinishBusy") then return end
    prompt:SetAttribute("__AutoFinishBusy", true)
    pcall(function() if prompt.InputHoldBegin then prompt:InputHoldBegin() end end)
    pcall(function() if prompt.HoldBegin then prompt:HoldBegin() end end)
    pcall(function() if prompt.Trigger then prompt:Trigger() end end)
    task.wait(duration)
    pcall(function() if prompt.InputHoldEnd then prompt:InputHoldEnd() end end)
    pcall(function() if prompt.HoldEnd then prompt:HoldEnd() end end)
    prompt:SetAttribute("__AutoFinishBusy", nil)
end

end

local FOV_Lines = {}
local FOV_Segment_Count = 48
pcall(function()
    for i = 1, FOV_Segment_Count do
        FOV_Lines[i] = Drawing.new("Line")
        FOV_Lines[i].Visible = false
        FOV_Lines[i].Thickness = 1.5
    end
end)

local function updateFOVLines(visible)
    if not visible then
        for i = 1, FOV_Segment_Count do
            if FOV_Lines[i] then FOV_Lines[i].Visible = false end
        end
        return
    end
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local step = (math.pi * 2) / FOV_Segment_Count
    for i = 1, FOV_Segment_Count do
        local a1 = (i - 1) * step
        local a2 = i * step
        if FOV_Lines[i] then
            FOV_Lines[i].From = center + Vector2.new(math.cos(a1) * H, math.sin(a1) * H)
            FOV_Lines[i].To = center + Vector2.new(math.cos(a2) * H, math.sin(a2) * H)
            
            local ratio = i / FOV_Segment_Count
            local r = math.floor(255 * ratio + 180 * (1 - ratio))
            local g = math.floor(100 * ratio + 0 * (1 - ratio))
            local b = math.floor(180 * ratio + 255 * (1 - ratio))
            
            FOV_Lines[i].Color = Color3.fromRGB(r, g, b)
            FOV_Lines[i].Visible = true
        end
    end
end

local skeletonESPEnabled = false
local skeletonConnections = {}
local skeletonJoints = {
    {"Head","UpperTorso"},{"UpperTorso","LowerTorso"},
    {"UpperTorso","LeftUpperArm"},{"LeftUpperArm","LeftLowerArm"},{"LeftLowerArm","LeftHand"},
    {"UpperTorso","RightUpperArm"},{"RightUpperArm","RightLowerArm"},{"RightLowerArm","RightHand"},
    {"LowerTorso","LeftUpperLeg"},{"LeftUpperLeg","LeftLowerLeg"},{"LeftLowerLeg","LeftFoot"},
    {"LowerTorso","RightUpperLeg"},{"RightUpperLeg","RightLowerLeg"},{"RightLowerLeg","RightFoot"}
}

local function createSkeleton(player)
    if skeletonConnections[player] then return end
    local lines = {}
    local conn = RunService.RenderStepped:Connect(function()
        if not skeletonESPEnabled or not player or not player.Character then
            for _, line in pairs(lines) do
                if line then line.Visible = false end
            end
            return
        end
        local char = player.Character
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChild("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then
            for _, line in pairs(lines) do
                if line then line.Visible = false end
            end
            return
        end
        local isTarget = (I == player)
        for i, joint in ipairs(skeletonJoints) do
            local p1 = char:FindFirstChild(joint[1])
            local p2 = char:FindFirstChild(joint[2])
            if not lines[i] then
                pcall(function()
                    lines[i] = Drawing.new("Line")
                    lines[i].Thickness = 1.8
                    lines[i].Transparency = 0.85
                end)
            end
            local line = lines[i]
            if line then
                line.Color = isTarget and Color3.fromRGB(170,0,255) or Color3.fromRGB(230,180,255)
                if p1 and p2 then
                    local pos1, on1 = Camera:WorldToViewportPoint(p1.Position)
                    local pos2, on2 = Camera:WorldToViewportPoint(p2.Position)
                    if on1 and on2 and pos1.Z > 0 and pos2.Z > 0 then
                        line.From = Vector2.new(pos1.X, pos1.Y)
                        line.To = Vector2.new(pos2.X, pos2.Y)
                        line.Visible = true
                    else
                        line.Visible = false
                    end
                else
                    line.Visible = false
                end
            end
        end
    end)
    skeletonConnections[player] = {conn = conn, lines = lines}
end

local function removeSkeleton(player)
    if skeletonConnections[player] then
        for _, line in pairs(skeletonConnections[player].lines) do
            pcall(function()
                if line and line.Remove then line:Remove() end
            end)
        end
        pcall(function() skeletonConnections[player].conn:Disconnect() end)
        skeletonConnections[player] = nil
    end
end

    if backpack then

    local char = LocalPlayer.Character
    if char then
        for _, tool in pairs(char:GetChildren()) do
            if tool:IsA("Tool") and tool:GetAttribute("HealthRestoreAmount") then
                return tool
            end
        end
    end
    return nil
end

local function useHealthTool(tool)
    if healBusy then return end

    lastHealTime = tick()

    task.wait()
    tool:Activate()
    stopAnimations()
    task.wait(0.25)
    if oldTool and oldTool.Parent == char then
        oldTool.Parent = LocalPlayer.Backpack
    end
    healBusy = false
end

    Callback = function(state)
        c().ItemDroppedGraber = state

            spawn(function()

                    return workspace:FindFirstChild("DroppedItems") or workspace:FindFirstChild("Items") or workspace
                end
                local function PickupItem(Item, char)
                    pcall(function()
                        local Network = require(ReplicatedStorage:FindFirstChild("Modules") and ReplicatedStorage.Modules.Core and ReplicatedStorage.Modules.Core.Net)
                        if Network then
                            Network.send("pickup_item", Item)
                        end
                    end)
                end
                local function StartItemGrabber(Radius)
                    local function ScanAndPickup()
                        local char = LocalPlayer.Character
                        if not char or not char.Parent then return end
                        local CharacterPosition = char:GetPivot().Position
                        local DroppedItemsContainer = GetDroppedItemsContainer()
                        for _, Item in ipairs(DroppedItemsContainer:GetChildren()) do
                            if Item:IsA("Model") then
                                local ItemPosition = Item:GetPivot().Position
                                local Distance = (ItemPosition - CharacterPosition).Magnitude
                                if Distance <= Radius then
                                    PickupItem(Item, char)
                                end
                            end
                        end
                    end
                    while c().ItemDroppedGraber do
                        pcall(ScanAndPickup)
                        task.wait(0.5)
                    end
                end
                StartItemGrabber(c().GrabRadius or 45)
            end)
        end

local function lockTool(tool)

    end

local function setupBackpack(bp)
    if not bp then return end
    for _, tool in ipairs(bp:GetChildren()) do
        lockTool(tool)
    end
    bp.ChildAdded:Connect(lockTool)
end

    local bp = LocalPlayer:FindFirstChildOfClass("Backpack")
    if bp then
        setupBackpack(bp)
    else
        LocalPlayer.ChildAdded:Connect(function(child)

                setupBackpack(child)
            end
        end)
    end
end

local function hookButton(btn)
    if not btn then return end
    if btn:FindFirstChild("UnlocksAtText") then
        btn.UnlocksAtText.Visible = false
    end
    if btn:FindFirstChild("EmoteName") then
        btn.EmoteName.Visible = true
    end
    CoreUI.on_click(btn, function()
        local hum = Char.get_hum()
        if not hum or hum.Health <= 0 then return end
        if EmotesUI.current_emote_playing.get() == btn then
            EmotesUI.current_emote_playing.set(nil)
        else
            EmotesUI.current_emote_playing.set(btn)
        end
        task.wait(0.12)
        EmotesUI.enabled.set(false)
    end)
    EmotesUI.current_emote_playing.hook(function(current)
        if btn:FindFirstChild("EmoteEquipped") then
            btn.EmoteEquipped.Visible = (current == btn)
        end
    end)
end

local function createDroppedItemBillboard(itemModel)
    local target = itemModel:FindFirstChild("PickUpZone") or itemModel:FindFirstChildWhichIsA("BasePart") or itemModel.PrimaryPart
    if not target then return nil end
    local info = getWeaponInfo(itemModel)
    local rarityColor = info and at[info.Rarity] or Color3.fromRGB(180,160,210)
    local displayName = info and info.Name or itemModel.Name
    local amount = itemModel:GetAttribute("Amount") or 1
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "DroppedESP_" .. itemModel.Name
    billboard.Adornee = target
    billboard.Size = UDim2.new(0, 100, 0, 20)
    billboard.StudsOffset = Vector3.new(0, 1.0, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = GuiContainer
    local main = Instance.new("Frame", billboard)
    main.Size = UDim2.new(1,0,1,0)
    main.BackgroundTransparency = 1
    local layout = Instance.new("UIListLayout", main)
    layout.FillDirection = Enum.FillDirection.Horizontal
    layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
    layout.VerticalAlignment = Enum.VerticalAlignment.Center
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Padding = UDim.new(0,3)
    local iconFrame = Instance.new("Frame", main)
    iconFrame.Size = UDim2.new(0,18,0,18)
    iconFrame.BackgroundColor3 = Color3.fromRGB(15,10,25)
    iconFrame.BackgroundTransparency = 0.15
    iconFrame.LayoutOrder = 1
    Instance.new("UICorner", iconFrame).CornerRadius = UDim.new(1,0)
    local stroke = Instance.new("UIStroke", iconFrame)
    stroke.Color = rarityColor; stroke.Thickness = 1.5
    local badge = Instance.new("Frame", main)
    badge.AutomaticSize = Enum.AutomaticSize.X
    badge.Size = UDim2.new(0,0,0,16)
    badge.BackgroundColor3 = Color3.fromRGB(10,5,15)
    badge.BackgroundTransparency = 0.25
    badge.LayoutOrder = 2
    Instance.new("UICorner", badge).CornerRadius = UDim.new(0.5,0)
    local pad = Instance.new("UIPadding", badge)
    pad.PaddingLeft = UDim.new(0,6); pad.PaddingRight = UDim.new(0,6)
    local label = Instance.new("TextLabel", badge)
    label.AutomaticSize = Enum.AutomaticSize.X
    label.Size = UDim2.new(0,0,1,0)
    label.BackgroundTransparency = 1
    label.Font = Enum.Font.SourceSansBold
    label.TextSize = 10
    label.TextColor3 = Color3.fromRGB(245,230,255)
    label.Text = (amount > 1) and (displayName .. " x" .. tostring(amount)) or displayName
    return billboard
end

local function cleanupDroppedItemESP()
    for model, gui in pairs(droppedESP_UI) do
        if not model or not model.Parent then
            if gui and gui.Parent then
                gui:Destroy()
            end
            droppedESP_UI[model] = nil
        end
    end
end

RunService.RenderStepped:Connect(function()
    cleanupDroppedItemESP()
    if not itemESPEnabled then
        for _, gui in pairs(droppedESP_UI) do
            if gui then gui.Enabled = false end
        end
        return
    end
    local dropped = Workspace:FindFirstChild("DroppedItems")
    if not dropped then return end
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    for _, model in ipairs(dropped:GetChildren()) do
        if model:IsA("Model") and model.Name ~= "Money" and not model:GetAttribute("Locked") then
            local target = model:FindFirstChild("PickUpZone") or model:FindFirstChildWhichIsA("BasePart") or model.PrimaryPart
            if target then
                local dist = (target.Position - root.Position).Magnitude
                if dist <= 800 then
                    if not droppedESP_UI[model] or not droppedESP_UI[model].Parent then
                        droppedESP_UI[model] = createDroppedItemBillboard(model)
                    else
                        droppedESP_UI[model].Enabled = true
                    end
                else
                    if droppedESP_UI[model] then
                        droppedESP_UI[model].Enabled = false
                    end
                end
            end
        end
    end
end)
