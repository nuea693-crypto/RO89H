local sendEventCounter = 0
local sendFuncCounter = 0

local function Send_Send(...)
    local args = { ... }
    if Counter and type(Counter.event) == "number" then
        Counter.event = Counter.event + 1
        pcall(function()
            if SendRemote then
                SendRemote:FireServer(Counter.event, unpack(args))
            end
        end)
    else
        sendEventCounter = sendEventCounter + 1
        pcall(function()
            if SendRemote then
                SendRemote:FireServer(sendEventCounter, unpack(args))
            end
        end)
    end
end

local function netGet(...)
    local args = { ... }
    for i, v in ipairs(args) do
        if typeof(v) == "Instance" then
            if v:IsA("Model") and #v:GetChildren() == 0 then
                local dropped = Workspace:FindFirstChild("DroppedItems")
                if dropped then
                    local model = dropped:FindFirstChildWhichIsA("Model")
                    if model then args[i] = model else return end
                else return end
            end
        end
    end
    if Counter and type(Counter.func) == "number" then
        Counter.func = Counter.func + 1
        local ok, result = pcall(function()
            local get = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("Get")
            return get:InvokeServer(Counter.func, unpack(args))
        end)
        return result
    else
        sendFuncCounter = sendFuncCounter + 1
        local ok, result = pcall(function()
            local get = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("Get")
            return get:InvokeServer(sendFuncCounter, unpack(args))
        end)
        return result
    end
end

local function getPing()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return 0.15 end
    local ns = pg:FindFirstChild("NetworkStats")
    if not ns then return 0.15 end
    local pl = ns:FindFirstChild("PingLabel")
    if not pl then return 0.15 end
    local num = tonumber(pl.Text:match("%d+"))
    if not num then return 0.15 end
    local p = num / 1000
    return (p < 0 or p > 2) and 0.15 or p
end

local function hasSafeZoneOrImmunity(character)
    if not character then return false end
    local hum = character:FindFirstChildOfClass("Humanoid")
    if hum then
        if hum:GetAttribute("SafeZone") or hum:GetAttribute("Immune") or hum:GetAttribute("SpawnProtected") or hum:GetAttribute("Invincible") then
            return true
        end
    end
    if character:FindFirstChildOfClass("ForceField") then
        return true
    end
    return false
end

local function getValidTargetCharacter(rawObj)
    if not rawObj then return nil end
    local char = nil
    if rawObj:IsA("Player") then

        char = rawObj

    if not char then return nil end

end

local function hasLineOfSight(targetPart, targetCharacter)
    local myHead = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Head")
    local origin = myHead and myHead.Position or Camera.CFrame.Position
    local dest = targetPart.Position
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    
    local ignoreList = { LocalPlayer.Character }
    if targetCharacter then
        table.insert(ignoreList, targetCharacter)
    end
    raycastParams.FilterDescendantsInstances = ignoreList
    
    local result = Workspace:Raycast(origin, dest - origin, raycastParams)
    if not result then
        return true
    end
    
    local hitInst = result.Instance
    if hitInst then
        local model = hitInst:FindFirstAncestorOfClass("Model")
        if model and (model:FindFirstChildOfClass("VehicleSeat") or model:FindFirstChildOfClass("Seat") or model.Name:lower():find("car") or model.Name:lower():find("vehicle")) then
            return true
        end
    end
    
    return false
end

local function getSmartAimPart(character)
    if not character then return nil end
    local head = character:FindFirstChild('Head')
    local hrp = character:FindFirstChild('HumanoidRootPart') or character:FindFirstChild('UpperTorso')

    if SilentAim.TargetMode == 'Torso' then
        return hrp or head
    elseif SilentAim.TargetMode == 'Random' then
        if math.random() <= 0.2 and head then
            return head
        else
            return hrp or head
        end
    else
        return head or hrp
    end
end

local previousTargetPositions = {}

local function getClosestTarget()
    local best, bestDist = nil, SilentAim.FOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

    local okChar, CharModule = pcall(function() return require(ReplicatedStorage.Modules.Core.Char) end)
    local candidates = {}

    if okChar and CharModule and CharModule.get_all then
        for _, raw in pairs(CharModule.get_all()) do
            local targetChar = getValidTargetCharacter(raw)
            if targetChar and targetChar ~= LocalPlayer.Character then
                local player = Players:GetPlayerFromCharacter(targetChar)
                if not player or not isWhitelisted(player) then
                    if not hasSafeZoneOrImmunity(targetChar) then
                        local hum = targetChar:FindFirstChildOfClass("Humanoid")
                        local targetPart = getSmartAimPart(targetChar)

                        if hum and hum.Health > 0 and targetPart then
                            local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                            if onScreen then
                                local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                                local dist3D = myRoot and (myRoot.Position - targetPart.Position).Magnitude or 0
                                table.insert(candidates, {
                                    target = player or { Character = targetChar },
                                    dist2D = dist2D,
                                    dist3D = dist3D
                                })
                            end
                        end
                    end
                end
            end
        end
    else
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and not isWhitelisted(player) and player.Character then
                local targetChar = getValidTargetCharacter(player.Character)
                if targetChar and not hasSafeZoneOrImmunity(targetChar) then
                    local hum = targetChar:FindFirstChildOfClass("Humanoid")
                    local targetPart = getSmartAimPart(targetChar)

                    if hum and hum.Health > 0 and targetPart then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                        if onScreen then
                            local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                            local dist3D = myRoot and (myRoot.Position - targetChar.HumanoidRootPart.Position).Magnitude or 0
                            table.insert(candidates, {
                                target = player,
                                dist2D = dist2D,
                                dist3D = dist3D
                            })
                        end
                    end
                end
            end
        end
    end

    table.sort(candidates, function(a, b)
        return a.dist2D < b.dist2D
    end)

    if #candidates > 0 then
        return candidates[1].target
    end
    return nil
end

local function getClosestEnemyToMe(maxRange)
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local best, bestDist = nil, maxRange or math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and not isWhitelisted(player) and player.Character then
            local targetChar = getValidTargetCharacter(player.Character)
            if targetChar and not hasSafeZoneOrImmunity(targetChar) then
                local hum  = targetChar:FindFirstChildOfClass("Humanoid")
                local root = targetChar:FindFirstChild("HumanoidRootPart")
                if hum and hum.Health > 0 and root then
                    local d = (root.Position - myRoot.Position).Magnitude
                    if d < bestDist then bestDist = d; best = player end
                end
            end
        end
    end
    return best
end

local function predictPosition(targetPart)
    if not targetPart then return Vector3.zero end
    local ping = getPing()
    local hrp = targetPart.Parent:FindFirstChild("HumanoidRootPart") or targetPart
    local currentPos = targetPart.Position
    
    local targetKey = targetPart.Parent
    local lastData = previousTargetPositions[targetKey]
    local frameVelocity = Vector3.zero
    
    if lastData then
        local timePassed = tick() - lastData.time
        if timePassed > 0 then
            frameVelocity = (currentPos - lastData.pos) / timePassed
        end
    end
    previousTargetPositions[targetKey] = { pos = currentPos, time = tick() }
    
    local vel = hrp.AssemblyLinearVelocity or hrp.Velocity or Vector3.zero
    local blendedVel = (vel * 0.5) + (frameVelocity * 0.5)
    
    if blendedVel.Magnitude > 350 then
        blendedVel = vel.Unit * 50
    end
    
    local adjustedVel = Vector3.new(blendedVel.X, 0, blendedVel.Z)
    local predicted = currentPos + (adjustedVel * ping * SilentAim.Prediction)
    return predicted
end

local function isShotgun(tool)
    local char = LocalPlayer.Character
    if not char then return false end
    local targetTool = tool
    if not targetTool then

            if t:IsA("Tool") then
                targetTool = t
                break
            end
        end
    end
    if targetTool then

        if ammo == "shotgun" or ammo == "shootgun" or targetTool.Name:lower():find("shotgun") or targetTool.Name:lower():find("spas") then
            return true
        end
    end
    return false
end

local function performAutoReload(tool)
    if not AutoReload.Enabled or not tool or not tool:IsA("Tool") then return end
    if tick() - AutoReload.LastReload < 0.8 then return end
    
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local char = LocalPlayer.Character
    if not backpack or not char then return end
    
    AutoReload.LastReload = tick()
    pcall(function()
        tool.Parent = backpack
        task.wait(0.03)
        tool.Parent = char
    end)
end

local function getClosestTargetForAutoShoot()
    local best, bestDist = nil, math.huge
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and not isWhitelisted(player) and player.Character then
            local targetChar = getValidTargetCharacter(player.Character)
            if targetChar and not hasSafeZoneOrImmunity(targetChar) then
                local hum = targetChar:FindFirstChildOfClass("Humanoid")
                local targetPart = getSmartAimPart(targetChar)

                if hum and hum.Health > 30 and targetPart then
                    if hasLineOfSight(targetPart, targetChar) then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                        if onScreen then
                            local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                            if dist < bestDist then
                                bestDist = dist
                                best = player
                            end
                        end
                    end
                end
            end
        end
    end
    return best
end

local function FindFireRateAttribute(gun)
    if not gun then return nil end

    for attrName, attrValue in pairs(gun:GetAttributes()) do
        if type(attrName) == 'string' and attrName:sub(-3) == '486' then
            return attrName
        end
    end
    return nil
end

    if not tool or not tool:IsA('Tool') then return false end
    return tool:GetAttribute('reload_time') 
        or tool:GetAttribute('AmmoType')

    if not gun or not gun:IsA('Tool') then return end

    if attr then
        gun:SetAttribute(attr, FIRE_RATE_VALUE)

        gun:SetAttribute('fire_rate', FIRE_RATE_VALUE)
    end
end

local function modifyFists(tool, enable)
    if not tool then return end
    local attrs = tool:GetAttributes()
    local keys  = {}
    for k in pairs(attrs) do table.insert(keys, k) end
    table.sort(keys)
    if #keys >= 7 then
        local rangeKey = keys[6]
        local dmgKey   = keys[7]
        if enable then
            if originalAttribs[rangeKey] == nil then originalAttribs[rangeKey] = tool:GetAttribute(rangeKey) end
            if originalAttribs[dmgKey]   == nil then originalAttribs[dmgKey]   = tool:GetAttribute(dmgKey)   end
            tool:SetAttribute(rangeKey, 360)
            tool:SetAttribute(dmgKey,   20)
        else
            if originalAttribs[rangeKey] then tool:SetAttribute(rangeKey, originalAttribs[rangeKey]) end
            if originalAttribs[dmgKey]   then tool:SetAttribute(dmgKey,   originalAttribs[dmgKey])   end
        end
    end
end

    for _, tool in ipairs(backpack:GetChildren()) do

local function getPlayersInRange(range)
    local result = {}
    local myChar = LocalPlayer.Character
    if not myChar or not myChar.PrimaryPart then return result end
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character.PrimaryPart then
            local ok, dist = pcall(function()
                return (player.Character.PrimaryPart.Position - myChar.PrimaryPart.Position).Magnitude
            end)
            if ok and dist and dist <= range then
                table.insert(result, player)
            end
        end
    end
    return result

local function attackNearby()
    if not SendRemote then return end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar.PrimaryPart then return end
    local tool = getActiveTool()
    if not tool or not isMeleeByName(tool) then return end
    local ok, parent = pcall(function() return tool.Parent end)
    if not ok or parent ~= LocalPlayer.Character then return end
    local nearby = getPlayersInRange(20)
    if #nearby == 0 then return end
    local ok2, myPos = pcall(function() return myChar.PrimaryPart.Position end)
    if not ok2 or not myPos then return end
    local targets, positions = {}, {}
    for _, player in pairs(nearby) do
        if player and player.Character and player.Character.PrimaryPart then
            local head = player.Character:FindFirstChild("Head")
            local root = player.Character.PrimaryPart
            if head and root then
                table.insert(targets, player)
                table.insert(positions, head.Position)
            end
        end
    end
    if #targets == 0 then return end
    local lookAt = CFrame.lookAt(myPos, positions[1])
    pcall(function()
        if Counter and type(Counter.event) == "number" then
            Counter.event = Counter.event + 1
        else
            sendEventCounter = sendEventCounter + 1
        end
        local ctr = (Counter and Counter.event) or sendEventCounter
        SendRemote:FireServer(ctr, "melee_attack", tool, targets, lookAt, 0.75)
    end)
end

local function UpdateJoystick(position)
    if not JoystickBase or not JoystickBase.Visible then return end
    local center = JoystickBase.AbsolutePosition + (JoystickBase.AbsoluteSize / 2)
    local radius = math.min(JoystickBase.AbsoluteSize.X, JoystickBase.AbsoluteSize.Y) / 2
    local delta = Vector2.new(position.X, position.Y) - center
    if delta.Magnitude > radius then delta = delta.Unit * radius end
    MobileX = math.clamp(delta.X / radius, -1, 1)
    MobileY = math.clamp(-delta.Y / radius, -1, 1)
    JoystickKnob.Position = UDim2.new(0.5, delta.X, 0.5, delta.Y)
end

local function DisableSelfCollision(primaryPart)
    savedCollisionParts = {}
    local carModel = primaryPart:FindFirstAncestorOfClass("Model")
    if carModel then
        for _, part in ipairs(carModel:GetDescendants()) do
            if part:IsA("BasePart") then
                savedCollisionParts[part] = part.CanCollide
                part.CanCollide = false
            end
        end
    else
        savedCollisionParts[primaryPart] = primaryPart.CanCollide
        primaryPart.CanCollide = false
    end
end

local function RestoreSelfCollision()
    for part, value in pairs(savedCollisionParts) do
        if part and part.Parent then
            part.CanCollide = value
        end
    end
    savedCollisionParts = {}
end

    if att1 then att1:Destroy() att1 = nil end

local function StartCarControl(seat)
    if not seat or not FlyingCarEnabled then return end
    StopCarControl()
    currentSeat = seat
    ShowJoystick(true)

    local primaryPart = seat.AssemblyRootPart or seat
    currentPrimary = primaryPart

    DisableSelfCollision(primaryPart)

    local groundY = primaryPart.Position.Y

    primaryPart.CustomPhysicalProperties = PhysicalProperties.new(0.7, 0.3, 0.5, 1, 1)

    att1 = Instance.new("Attachment") att1.Parent = primaryPart
    att2 = Instance.new("Attachment") att2.Parent = primaryPart
    att3 = Instance.new("Attachment") att3.Parent = primaryPart

    alignOrientation = Instance.new("AlignOrientation")
    alignOrientation.Attachment0 = att1
    alignOrientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
    alignOrientation.MaxTorque = 500000
    alignOrientation.Responsiveness = 100
    alignOrientation.RigidityEnabled = true
    alignOrientation.PrimaryAxisOnly = false
    alignOrientation.CFrame = CFrame.new(primaryPart.Position)
        * CFrame.Angles(0, math.atan2(-primaryPart.CFrame.LookVector.X, -primaryPart.CFrame.LookVector.Z), 0)
    alignOrientation.Parent = primaryPart

    linearVelocity = Instance.new("LinearVelocity")
    linearVelocity.Attachment0 = att2
    linearVelocity.MaxForce = 100000
    linearVelocity.VectorVelocity = Vector3.new(0, 0, 0)
    linearVelocity.ForceLimitMode = Enum.ForceLimitMode.PerAxis
    linearVelocity.MaxAxesForce = Vector3.new(100000, 100000, 100000)
    linearVelocity.Parent = primaryPart

    antiGravityForce = Instance.new("VectorForce")
    antiGravityForce.Attachment0 = att3
    antiGravityForce.Force = Vector3.new(0, primaryPart.AssemblyMass * Workspace.Gravity, 0)
    antiGravityForce.RelativeTo = Enum.ActuatorRelativeTo.World
    antiGravityForce.Parent = primaryPart

    local currentYaw = math.atan2(-primaryPart.CFrame.LookVector.X, -primaryPart.CFrame.LookVector.Z)
    local flipCheckCooldown = 0
    local currentVelocity = Vector3.new(0, 0, 0)

    controlConnection = RunService.Heartbeat:Connect(function(dt)
        if not FlyingCarEnabled then StopCarControl() return end
        if not primaryPart or not primaryPart.Parent then StopCarControl() return end
        if not currentSeat or not currentSeat.Parent then StopCarControl() return end

        local targetY = groundY + HoverHeight

        local rotateInput = 0
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then rotateInput = rotateInput + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then rotateInput = rotateInput - 1 end
        if math.abs(MobileX) > 0.05 then rotateInput = MobileX end
        if rotateInput ~= 0 then
            currentYaw = currentYaw - rotateInput * CarRotateSpeed * dt
        end

        local pos = primaryPart.Position
        local newCFrame = CFrame.new(pos.X, targetY, pos.Z) * CFrame.Angles(0, currentYaw, 0)
        alignOrientation.CFrame = newCFrame

        local moveInput = 0
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveInput = moveInput + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveInput = moveInput - 1 end
        if math.abs(MobileY) > 0.05 then moveInput = MobileY end

        local forward = Vector3.new(newCFrame.LookVector.X, 0, newCFrame.LookVector.Z).Unit

        local targetVelocity
        if math.abs(moveInput) > 0.05 then
            local speedMultiplier
            if moveInput < 0 then
                speedMultiplier = CarSpeed * 0.4
            else
                speedMultiplier = CarSpeed
            end
            targetVelocity = forward * moveInput * speedMultiplier
        else
            targetVelocity = Vector3.new(0, 0, 0)
        end

        local lerpAlpha = (math.abs(moveInput) > 0.05) and Acceleration or Deceleration
        currentVelocity = currentVelocity:Lerp(targetVelocity, lerpAlpha)

        if currentVelocity.Magnitude < 0.3 then
            currentVelocity = Vector3.new(0, 0, 0)
        end

        linearVelocity.VectorVelocity = Vector3.new(
            currentVelocity.X,
            0,
            currentVelocity.Z
        )

        if math.abs(primaryPart.Position.Y - targetY) > 1.5 then
            primaryPart.CFrame = CFrame.new(primaryPart.Position.X, targetY, primaryPart.Position.Z)
                * CFrame.Angles(0, currentYaw, 0)
            primaryPart.AssemblyLinearVelocity = Vector3.new(
                primaryPart.AssemblyLinearVelocity.X,
                0,
                primaryPart.AssemblyLinearVelocity.Z
            )
        end

        flipCheckCooldown = flipCheckCooldown + dt
        if flipCheckCooldown >= 0.1 then
            flipCheckCooldown = 0
            if math.abs(primaryPart.CFrame.UpVector.Y) < 0.7 then
                primaryPart.CFrame = CFrame.new(pos.X, targetY, pos.Z) * CFrame.Angles(0, currentYaw, 0)
                primaryPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                primaryPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                currentVelocity = Vector3.new(0, 0, 0)
            end
        end
    end)
end

_G.BumpAura = false
_G.BumpAuraThread = nil
local BumpLastHit = {}
local BUMP_RADIUS = 350
local BUMP_COOLDOWN = 0.08
local BUMP_LOOP_WAIT = 0.03
local BUMP_FORCE = 1500

local function IsAliveChar(char)
    if not char or not char.Parent then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    return char:FindFirstChild("HumanoidRootPart") ~= nil
end

local function GetMyCarAndChassis()
    local ok, VehicleModule = pcall(function() return require(ReplicatedStorage.Modules.Game.VehicleSystem.Vehicle) end)
    if not ok or not VehicleModule then return nil, nil end
    local car = VehicleModule.get_car_player_is_in()
    if not car then return nil, nil end
    local chassis = car:FindFirstChild("Chassis")
        or car.PrimaryPart
        or car:FindFirstChildWhichIsA("BasePart")
    return car, chassis
end

local function BumpAuraLoop()
    while _G.BumpAura do
        task.wait(BUMP_LOOP_WAIT)
        if not _G.BumpAura then break end

        local car, chassis = GetMyCarAndChassis()
        if car and chassis then
            for k in pairs(BumpLastHit) do
                if not k.Parent then BumpLastHit[k] = nil end
            end

            pcall(function()
                local okChar, CharModule = pcall(function() return require(ReplicatedStorage.Modules.Core.Char) end)
                if not okChar or not CharModule then return end

                for _, raw in pairs(CharModule.get_all()) do
                    if not _G.BumpAura then break end
                    local targetChar = getValidTargetCharacter(raw)
                    if targetChar and targetChar ~= LocalPlayer.Character and IsAliveChar(targetChar) then
                        if not hasSafeZoneOrImmunity(targetChar) then
                            local hrp = targetChar:FindFirstChild("HumanoidRootPart")
                            local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                            if hrp and myRoot then
                                local dist = (myRoot.Position - hrp.Position).Magnitude
                                if dist <= BUMP_RADIUS then
                                    local now = os.clock()
                                    if now - (BumpLastHit[targetChar] or 0) >= BUMP_COOLDOWN then
                                        BumpLastHit[targetChar] = now
                                        local vel = chassis.AssemblyLinearVelocity
                                        local force = (vel.Magnitude < 5) and ((hrp.Position - chassis.Position).Unit * BUMP_FORCE) or (vel + Vector3.new(0, BUMP_FORCE * 0.1, 0))
                                        Send_Send("run_over", car, targetChar, force)
                                    end
                                end
                            end
                        end
                    end
                end
            end)
        end
    end
end

            _G.BumpAuraThread = task.spawn(function()

MiscTab:Toggle({
    Title = "Frustum Culling (แก้กล้องไม่ตาม Object / ซ่อนวัตถุนอกจอ)",
    Icon = "eye",
    Default = false,
    Callback = function(state)
        if state then
            if frustumCullingConnection then frustumCullingConnection:Disconnect() end
            frustumCullingConnection = RunService.RenderStepped:Connect(function()
                local cam = Workspace.CurrentCamera
                if not cam then return end
                for _, obj in ipairs(Workspace:GetDescendants()) do
                    if obj:IsA("BasePart") and not obj:IsDescendantOf(LocalPlayer.Character or workspace) then
                        local _, onScreen = cam:WorldToViewportPoint(obj.Position)
                        if onScreen then
                            if obj:GetAttribute("OriginalTransparency") == nil then
                                obj:SetAttribute("OriginalTransparency", obj.Transparency)
                            end
                            obj.Transparency = obj:GetAttribute("OriginalTransparency") or 0
                        else
                            if obj:GetAttribute("OriginalTransparency") == nil then
                                obj:SetAttribute("OriginalTransparency", obj.Transparency)
                            end
                            if not obj.Anchored and obj.Size.Magnitude > 10 then
                                obj.Transparency = 0.95
                            end
                        end
                    end
                end
            end)
            if WindUI and WindUI.Notify then
                WindUI:Notify({ Title = "NYVEX HUB", Content = "✅ เปิดใช้งาน Frustum Culling เรียบร้อย!", Duration = 2 })
            end
        else
            if frustumCullingConnection then
                frustumCullingConnection:Disconnect()
                frustumCullingConnection = nil
            end
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("BasePart") then
                    local orig = obj:GetAttribute("OriginalTransparency")
                    if orig then
                        obj.Transparency = orig
                    end
                end
            end
            if WindUI and WindUI.Notify then
                WindUI:Notify({ Title = "NYVEX HUB", Content = "❌ ปิดใช้งาน Frustum Culling แล้ว", Duration = 2 })
            end
        end
    end
})
