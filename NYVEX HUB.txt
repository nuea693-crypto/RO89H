​local function getPing()
local pg = LocalPlayer:FindFirstChild("PlayerGui")
if not pg then return 0.15 end
local ns = pg:FindFirstChild("NetworkStats")
if not ns then return 0.15 end
local pl = ns:FindFirstChild("PingLabel")
if not pl then return 0.15 end
local num = tonumber(pl.Text:match("%d+"))
if not num then return 0.15 end
local p = num / 1000
return (p < 0 or p > 2) and 0.15 or p
end

​local function getSmartAimPart(character)
if not character then return nil end
local head = character:FindFirstChild('Head')
local hrp = character:FindFirstChild('HumanoidRootPart')
if SilentAim.TargetMode == 'Torso' then
return hrp or head
elseif SilentAim.TargetMode == 'Smart' then
if head and hrp then
local vel = hrp.AssemblyLinearVelocity or hrp.Velocity or Vector3.zero
if vel.Magnitude > 25 then return hrp else return head end
end
return head or hrp
else
return head or hrp
end
end
​/* STREAMING_CHUNK:Defining targeting and 100% hit calculations... */
local function getClosestTarget()
local best, bestDist = nil, math.huge
local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
​for _, player in ipairs(Players:GetPlayers()) do
if player ~= LocalPlayer and not isWhitelisted(player) and player.Character then
local hum = player.Character:FindFirstChildOfClass("Humanoid")
local targetPart = getSmartAimPart(player.Character)
​if targetPart then
local isAlive = (hum and hum.Health > 0) or true
if isAlive then
local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
local dist = onScreen and (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and (player.Character.HumanoidRootPart.Position - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude) or 0
if dist <= bestDist then
bestDist = dist
best = player
end
end
end
end
end
return best
end
​local function getClosestEnemyToMe(maxRange)
local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
if not myRoot then return nil end
local best, bestDist = nil, maxRange or math.huge
for _, player in ipairs(Players:GetPlayers()) do
if player ~= LocalPlayer and not isWhitelisted(player) and player.Character then
local root = player.Character:FindFirstChild("HumanoidRootPart")
if root then
local d = (root.Position - myRoot.Position).Magnitude
if d < bestDist then bestDist = d; best = player end
end
end
end
return best
end
​local function predictPosition(targetPart)
if not targetPart then return Vector3.zero end
local ping = getPing()
local hrp = targetPart.Parent:FindFirstChild("HumanoidRootPart") or targetPart
local vel = hrp.AssemblyLinearVelocity or hrp.Velocity or Vector3.zero
​local adjustedVel = Vector3.new(vel.X, 0, vel.Z)
local predicted = targetPart.Position + (adjustedVel * ping * SilentAim.Prediction)
return predicted
end

​local function getCrateOptions()
local map = workspace:FindFirstChild("Map")
if not map then return nil end
local tiles = map:FindFirstChild("Tiles")
if not tiles then return nil end
local gunShopTile = tiles:FindFirstChild("GunShopTile")
if not gunShopTile then return nil end
local patriotWeapons = gunShopTile:FindFirstChild("PatriotWeapons")
if not patriotWeapons then return nil end
local interior = patriotWeapons:FindFirstChild("Interior")
if not interior then return nil end
local crates = interior:FindFirstChild("Crates")
if not crates then return nil end
local ammoCrate = crates:FindFirstChild("Ammo Crate")
if not ammoCrate then return nil end
return ammoCrate:FindFirstChild("CrateOptions")
end
​local function openCrateWithType(bulletType)
local crateOptions = getCrateOptions()
if not crateOptions then
warn("❌ ไม่พบ CrateOptions")
return
end
local targetItem = crateOptions:FindFirstChild(bulletType)
if not targetItem then return end
netGet("open_crate", targetItem, "money")
end
