local function getPing()
    local gui = LocalPlayer:FindFirstChild("PlayerGui")
    if not gui then return 0.2 end
    local stats = gui:FindFirstChild("NetworkStats")
    if not stats then return 0.2 end
    local label = stats:FindFirstChild("PingLabel")
    if not label then return 0.2 end
    local num = tonumber(tostring(label.Text):match("%d+"))
    if not num then return 0.2 end
    local ping = num / 1000
    return (ping < 0 or ping > 2) and 0.2 or ping
end

local function isPlayerExcluded(name)
    for _, entry in ipairs(excludedPlayers) do
        if entry ~= "" and string.find(string.lower(name), string.lower(entry)) then
            return true
        end
    end
    return false
end

local function safeCall(fn, ...)
    local ok, result = pcall(fn, ...)
    return ok, result
end

local unpackTable = table.unpack or unpack

local function callRemote(remote, ...)
    if not remote then return end
    local args = { ... }

    if remote.ClassName == "RemoteEvent" then
        if Counter and type(Counter.event) == "number" then
            Counter.event = Counter.event + 1
            safeCall(function(...) remote:FireServer(Counter.event, ...) end, unpackTable(args))
        else
            sendEventCounter = sendEventCounter + 1
            safeCall(function(...) remote:FireServer(sendEventCounter, ...) end, unpackTable(args))
        end
    elseif remote.ClassName == "RemoteFunction" then
        if Counter and type(Counter.func) == "number" then
            Counter.func = Counter.func + 1
            safeCall(function(...) remote:InvokeServer(Counter.func, ...) end, unpackTable(args))
        else
            sendFuncCounter = sendFuncCounter + 1
            safeCall(function(...) remote:InvokeServer(sendFuncCounter, ...) end, unpackTable(args))
        end
    else
        safeCall(function(...)
            if remote.FireServer   then remote:FireServer(...)   end
            if remote.InvokeServer then remote:InvokeServer(...) end
        end, unpackTable(args))
    end
end

local function netGet(...)
    if not Counter or not Counter.func then return end
    local args = { ... }
    for i, v in ipairs(args) do
        if typeof(v) == "Instance" then
            if v:IsA("Model") and #v:GetChildren() == 0 then
                local dropped = Workspace:FindFirstChild("DroppedItems")
                if dropped then
                    local model = dropped:FindFirstChildWhichIsA("Model")
                    if model then args[i] = model else return end
                else return end
            end
        end
    end
    Counter.func = (Counter.func or 0) + 1
    local ok, result = pcall(function()
        local get = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("Get")
        return get:InvokeServer(Counter.func, unpackTable(args))
    end)
    if not ok then warn("[NetGet Error]", result) end
    return result
end

end

local HISTORY_SIZE    = 10
local VELOCITY_SCALE  = 1.25
local MAX_JUMP_VEL    = 150
local SMOOTH_MULT     = 1

                positionHistory[player] = positionHistory[player] or {}
                table.insert(positionHistory[player], { time = os.clock(), pos = root.Position })

                end
            else

            end
        end
    end

local function calculateVelocity(player)
    local history = positionHistory[player]
    if not history or #history < 2 then return Vector3.new() end
    local sum, totalWeight = Vector3.new(), 0
    for i = 2, #history do
        local dt = history[i].time - history[i - 1].time
        if dt > 0 then
            local raw = (history[i].pos - history[i - 1].pos) / dt
            local clamped = Vector3.new(
                math.clamp(raw.X, -120, 120),
                math.clamp(raw.Y, -150, 150),
                math.clamp(raw.Z, -120, 120)
            )
            local weight = i
            sum         = sum + clamped * weight
            totalWeight = totalWeight + weight
        end
    end
    if totalWeight == 0 then return Vector3.new() end
    local avg = sum / totalWeight
    if avg.Y > MAX_JUMP_VEL then
        return Vector3.new(avg.X * 1.15, math.clamp(avg.Y * 0.85, 0, 400), avg.Z * 1.15)
    end
    return avg
end

local function predictPosition(part, root)
    if not part then return Vector3.zero end

    local parentModel = part.Parent
    local player = parentModel and Players:GetPlayerFromCharacter(parentModel)
    local velocity = (player and calculateVelocity(player)) or Vector3.zero

    local ping = math.clamp(getPing(), 0.06, 0.35)
    local hSpeed = Vector3.new(velocity.X, 0, velocity.Z).Magnitude

    local multiplier
    if     hSpeed > 60 then multiplier = 1.50
    elseif hSpeed > 50 then multiplier = 1.42
    elseif hSpeed > 35 then multiplier = 1.32
    elseif hSpeed > 20 then multiplier = 1.22
    elseif hSpeed > 10 then multiplier = 1.15
    else                     multiplier = 1.05
    end

    if ping > 0.15 then
        multiplier = multiplier * 0.93
    end

    local horizontal = Vector3.new(velocity.X, 0, velocity.Z) * ping * multiplier

    local vertical = Vector3.new(
        0,
        math.clamp(velocity.Y * ping * 0.30, -4, 4),
        0
    )

    local jumpBoost = Vector3.new(
        0,
        velocity.Y > 20 and 0.50
        or velocity.Y > 15 and 0.35
        or 0,
        0
    )

    local headOffset = Vector3.zero
    if part.Name == "Head" then
        headOffset = Vector3.new(
            0,
            hSpeed > 30 and 0.14
            or hSpeed > 22 and 0.10
            or 0.05,
            0
        )
    end

    local rawPos = part.Position
        + horizontal
        + vertical
        + jumpBoost
        + headOffset

    if rawPos.Y < 1 then
        rawPos = Vector3.new(rawPos.X, 1.5, rawPos.Z)
    end

    return rawPos
end

local function isBehindWall(origin, target)
    if not origin or not target then return false end
    local dir = target - origin
    if dir.Magnitude < 1 then return false end
    local params = RaycastParams.new()
    local result = Workspace:Raycast(origin, dir, params)
    if not result then return false end
    local inst = result.Instance
    local myChar    = LocalPlayer.Character
    local aimChar   = aimTarget and aimTarget.Character
    return inst and not (
        (myChar and inst:IsDescendantOf(myChar)) or
        (aimChar and inst:IsDescendantOf(aimChar))
    )
end

local function getClosestTarget()
    local best, bestDist = nil, fovRadius
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local head = player.Character:FindFirstChild("Head")
            local hum  = player.Character:FindFirstChild("Humanoid")
            local root = player.Character:FindFirstChild("HumanoidRootPart")
            if head and hum and hum.Health > 0 and root then
                local pos, visible = Camera:WorldToViewportPoint(head.Position)
                if visible then
                    local screenPos = Vector2.new(pos.X, pos.Y)
                    local dist = (screenPos - center).Magnitude
                    if dist <= fovRadius and not isPlayerExcluded(player.Name) then
                        if dist < bestDist then
                            bestDist = dist
                            best = player
                        end
                    end
                end
            end
        end
    end
    return best
end

local function isDowned()
    local ok, CharModule = pcall(function() return require(ReplicatedStorage.Modules.Core.Char) end)
    if not ok or not CharModule then return false end
    local hum = CharModule.get_hum()
    if not hum or hum.Health <= 0 then return false end

end

    local ok, CharModule = pcall(function() return require(ReplicatedStorage.Modules.Core.Char) end)

    local root = getHRP()
    if not root then return end

    root.CFrame = underMapPos
end

local function flickerAndMove()
    if isFlickering then return end
    isFlickering = true
    task.spawn(function()
        while isFlickering and antiKillEnabled and isDowned() do
            local ok, CharModule = pcall(function() return require(ReplicatedStorage.Modules.Core.Char) end)
            local hum = ok and CharModule.get_hum()
            if hum and hum.Health <= 0 then break end
            local root = getHRP()

                local offset = Vector3.new(math.cos(angle), 0, math.sin(angle)) * 10
                root.CFrame = CFrame.new(underMapPos.Position + offset)
                task.wait(0.05)
                root.CFrame = underMapPos
            end
            task.wait(0.1)
        end
        isFlickering = false
    end)
end

local function getPlayersInRange(range)
    local result = {}
    local myChar = LocalPlayer.Character
    if not myChar or not myChar.PrimaryPart then return result end
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character.PrimaryPart then

                return (player.Character.PrimaryPart.Position - myChar.PrimaryPart.Position).Magnitude
            end)

local function getActiveTool()
    local char = LocalPlayer.Character
    if char then
        for _, v in ipairs(char:GetChildren()) do
            if pcall(function() return v:IsA("Tool") end) and v:IsA("Tool") then
                return v
            end
        end
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, v in ipairs(backpack:GetChildren()) do
            if pcall(function() return v:IsA("Tool") end) and v:IsA("Tool") then
                return v
            end
        end
    end
    return nil
end

local function isMeleeToolCheck(tool)
    if not tool then return false end
    if tool.Name == "Fists" then return true end
    local itemsFolder = ReplicatedStorage:FindFirstChild("Items")
    if not itemsFolder then return false end
    local meleeFolder    = itemsFolder:FindFirstChild("melee")
    local throwableFolder = itemsFolder:FindFirstChild("throwable")
    if not meleeFolder or not throwableFolder then return false end
    return meleeFolder:FindFirstChild(tool.Name) and not throwableFolder:FindFirstChild(tool.Name)
end

local function attackNearby()
    if not SendRemote then return end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar.PrimaryPart then return end
    local tool = getActiveTool()
    if not tool or not isMeleeToolCheck(tool) then return end
    local ok, parent = pcall(function() return tool.Parent end)
    if not ok or parent ~= LocalPlayer.Character then return end
    local nearby = getPlayersInRange(20)
    if #nearby == 0 then return end
    local ok2, myPos = pcall(function() return myChar.PrimaryPart.Position end)
    if not ok2 or not myPos then return end
    local targets, positions = {}, {}
    for _, player in pairs(nearby) do
        if player and player.Character and player.Character.PrimaryPart then
            local head = player.Character:FindFirstChild("Head")
            local root = player.Character.PrimaryPart
            if head and root then
                local aimPos = predictPosition(head, root)
                table.insert(targets, player)
                table.insert(positions, aimPos)
            end
        end
    end
    if #targets == 0 then return end
    local lookAt = CFrame.lookAt(myPos, positions[1])
    pcall(function()
        callRemote(SendRemote, unpackTable({ "melee_attack", tool, targets, lookAt, 0.75 }))
    end)
end

    if autoAttackRunning then return end

            task.wait(0.4)
            if autoAttackEnabled and LocalPlayer.Character and LocalPlayer.Character.PrimaryPart then

    end)
end

-- ══════════════════════════════════════════════════════════════

-- ══════════════════════════════════════════════════════════════
local function setFinishPrompt(prompt)
    if prompt and prompt:IsA("ProximityPrompt") then
        prompt.HoldDuration = 1
        prompt.MaxActivationDistance = 20
    end
end

local function tryHoldPrompt(prompt, duration)
    if not prompt or prompt:GetAttribute("__AutoFinishBusy") then return end
    prompt:SetAttribute("__AutoFinishBusy", true)
    pcall(function() if prompt.InputHoldBegin then prompt:InputHoldBegin() end end)
    pcall(function() if prompt.HoldBegin      then prompt:HoldBegin()      end end)
    pcall(function() if prompt.Trigger        then prompt:Trigger()        end end)
    task.wait(duration)
    pcall(function() if prompt.InputHoldEnd then prompt:InputHoldEnd() end end)
    pcall(function() if prompt.HoldEnd      then prompt:HoldEnd()      end end)
    prompt:SetAttribute("__AutoFinishBusy", nil)
end

local function findFinishPrompts()
    local result = {}
    for _, child in pairs(Workspace:GetChildren()) do
        local player = Players:GetPlayerFromCharacter(child)
        if player and not isPlayerExcluded(player.Name) then
            local root = child:FindFirstChild("HumanoidRootPart")
            if root then
                local prompt = root:FindFirstChild("FinishPrompt")
                if prompt then
                    setFinishPrompt(prompt)
                    table.insert(result, prompt)
                end
            end
        end
    end
    return result
end

    end
end

                else

                end

                local topPos    = head.Position + Vector3.new(0, head.Size.Y / 2, 0)
                local bottomPos = root.Position - Vector3.new(0, 3.1, 0)

                local posHead, visHead = Camera:WorldToViewportPoint(topPos)
                local posFoot, visFoot = Camera:WorldToViewportPoint(bottomPos)

                local dist3D = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    and (LocalPlayer.Character.HumanoidRootPart.Position - root.Position).Magnitude or 0
                local scaleFactor = math.clamp(20 / math.max(dist3D, 1), 0.4, 1.2)
                local nameSize    = math.clamp(math.floor(15 * scaleFactor), 9, 17)
                local infoSize    = math.clamp(math.floor(14 * scaleFactor), 9, 16)

                if nameESPEnabled and visHead then
                    data.Name.Visible  = true
                    data.Name.Text     = player.Name
                    data.Name.Size     = nameSize
                    data.Name.Position = Vector2.new(posHead.X, posHead.Y - nameSize - 2)

                else
                    data.Name.Visible = false
                end

                if distanceESPEnabled and visFoot and dist3D > 0 then
                    local pct = math.floor(hum.Health / math.max(hum.MaxHealth, 1) * 100)
                    data.Info.Visible  = true
                    data.Info.Size     = infoSize
                    data.Info.Text     = pct .. "% | " .. math.floor(dist3D) .. "M"
                    data.Info.Position = Vector2.new(posFoot.X, posFoot.Y + 3)
                    data.Info.Color    = getESPColor(player)
                else
                    data.Info.Visible = false
                end
            else
                data.Highlight.Enabled = false
                data.Name.Visible      = false
                data.Info.Visible      = false
            end
        end
    end
end)

local _ToolHashes         = {}

local inventoryConnections = {}

local INV_ValidCategories = { gun=true, melee=true, accessory=true, consumable=true, throwable=true }

local INV_RARITY_COLORS = {
    ["Common"]    = Color3.fromRGB(255,255,255),
    ["UnCommon"]  = Color3.fromRGB(200,200,200),
    ["Rare"]      = Color3.fromRGB(160,160,160),
    ["Legendary"] = Color3.fromRGB(120,120,120),
    ["Epic"]      = Color3.fromRGB(80,80,80),
    ["Omega"]     = Color3.fromRGB(255,255,255),
    ["_default"]  = Color3.fromRGB(255,255,255),
}

local function INV_RarityColor(r)
    return INV_RARITY_COLORS[r] or INV_RARITY_COLORS["_default"]
end

local function INV_GenerateHash(Tool)
    local ids = {}
    for _, d in next, Tool:GetDescendants() do
        if d:IsA("MeshPart") then
            if d.MeshId ~= "" then table.insert(ids, d.MeshId) end
            if d.TextureID ~= "" then table.insert(ids, d.TextureID) end
            table.insert(ids, d.Material.Name)
        elseif d:IsA("SpecialMesh") then
            if d.MeshId ~= "" then table.insert(ids, d.MeshId) end
            if d.TextureId ~= "" then table.insert(ids, d.TextureId) end
            if d.Parent and d.Parent:IsA("BasePart") then table.insert(ids, d.Parent.Material.Name) end
        end
    end
    table.sort(ids, function(a, b)
        local na, nb = tonumber(a:match("%d+")), tonumber(b:match("%d+"))
        if na and nb then return na < nb end
        return a < b
    end)
    local s = table.concat(ids, ",")
    local h = 0
    for i = 1, #s do h = (h * 31 + s:byte(i)) % 2^32 end
    return h
end

local function INV_ResolveData(Tool)
    local cached = _resolvedToolCache[Tool]
    if cached then return cached end
    local ref = _ToolHashes[INV_GenerateHash(Tool)]
    if not ref then return nil end
    local result = { Name=ref.Name, Rarity=ref:GetAttribute("RarityName"), ImageId=ref:GetAttribute("ImageId"), Price=ref:GetAttribute("RarityPrice") }
    _resolvedToolCache[Tool] = result
    Tool.AncestryChanged:Connect(function()
        if not Tool.Parent then _resolvedToolCache[Tool] = nil end
    end)
    return result
end

local function INV_SetupPlayer(player)
    if player == LocalPlayer then return end
    local conns = {}
    local function onChanged()
        if inventoryESPEnabled then INV_BuildGUI(player) end
    end
    local bp = player:FindFirstChild("Backpack")
    if bp then
        table.insert(conns, bp.ChildAdded:Connect(onChanged))
        table.insert(conns, bp.ChildRemoved:Connect(onChanged))
    end
    table.insert(conns, player.CharacterAdded:Connect(function(char)
        bp = player:FindFirstChild("Backpack")
        if bp then
            table.insert(conns, bp.ChildAdded:Connect(onChanged))
            table.insert(conns, bp.ChildRemoved:Connect(onChanged))
        end
        char.ChildAdded:Connect(onChanged)
        char.ChildRemoved:Connect(onChanged)
        if inventoryESPEnabled then task.wait(); INV_BuildGUI(player) end
    end))
    if player.Character then
        local char = player.Character
        table.insert(conns, char.ChildAdded:Connect(onChanged))
        table.insert(conns, char.ChildRemoved:Connect(onChanged))
    end
    inventoryConnections[player] = conns
end

local function INV_TeardownPlayer(player)
    local conns = inventoryConnections[player]
    if conns then for _, c in pairs(conns) do c:Disconnect() end end
    inventoryConnections[player] = nil
    INV_DestroyGUI(player)
end

local function FindFireRateAttribute(gun)
    if not gun then return nil end
    if gun:GetAttribute('fire_rate') ~= nil then
        return 'fire_rate'
    end
    for attrName, attrValue in pairs(gun:GetAttributes()) do
        if type(attrName) == 'string' and attrName:sub(-3) == '486' then
            return attrName
        end
    end
    return nil
end

local function isGunTool(tool)
    if not tool or not tool:IsA("Tool") then return false end
    return (GunItems and GunItems:FindFirstChild(tool.Name) ~= nil)
        or tool.Name:match("Gun")
        or tool:FindFirstChild("Handle")
        or FindFireRateAttribute(tool) ~= nil
end

local function applyGodGun(tool)
    if not tool or not isGunTool(tool) then return end
    pcall(function()
        tool:SetAttribute("accuracy",   getEnv().AccuracyValue)
        tool:SetAttribute("Recoil",     getEnv().RecoilValue)
        tool:SetAttribute("Durability", getEnv().Durability)
        tool:SetAttribute("automatic",  getEnv().AutoValue)
        
        if getEnv().InfiniteFireRate then
            local attr = FindFireRateAttribute(tool)
            if attr then
                tool:SetAttribute(attr, FIRE_RATE_MAX)
            else
                tool:SetAttribute("fire_rate", FIRE_RATE_MAX)
            end
        end
    end)
end

    for _, name in ipairs(meleeNames) do
        if tool.Name == name then return true end

local function modifyFists(tool, enable)
    if not tool then return end
    local attrs = tool:GetAttributes()
    local keys  = {}
    for k in pairs(attrs) do table.insert(keys, k) end
    table.sort(keys)
    if #keys >= 7 then
        local rangeKey = keys[6]
        local dmgKey   = keys[7]
        if enable then
            if originalAttribs[rangeKey] == nil then originalAttribs[rangeKey] = tool:GetAttribute(rangeKey) end
            if originalAttribs[dmgKey]   == nil then originalAttribs[dmgKey]   = tool:GetAttribute(dmgKey)   end
            tool:SetAttribute(rangeKey, 360)
            tool:SetAttribute(dmgKey,   20)
        else
            if originalAttribs[rangeKey] then tool:SetAttribute(rangeKey, originalAttribs[rangeKey]) end
            if originalAttribs[dmgKey]   then tool:SetAttribute(dmgKey,   originalAttribs[dmgKey])   end
        end
    end
end

    for _, tool in ipairs(char:GetChildren()) do

    end
    for _, tool in ipairs(backpack:GetChildren()) do

    end
end

local _jumpFlySpeed = 40
local _jumpConns = {}

local function stopHighJump()
    for _, c in ipairs(_jumpConns) do
        pcall(function() c:Disconnect() end)
    end
    _jumpConns = {}
    _jumpHolding = false
    pcall(function() ContextActionService:UnbindAction("_HJ_FlyUp") end)
    local char = LocalPlayer.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            root.Velocity = Vector3.new(root.Velocity.X, 0, root.Velocity.Z)
        end
    end
end

local function setupHighJump(char)
    if not jumpPowerEnabled then return end
    stopHighJump()
    local humanoid = char:WaitForChild("Humanoid")
    local hrp = char:WaitForChild("HumanoidRootPart")

    local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")
    local function hookJumpButton()
        local screenGui = PlayerGui:FindFirstChild("TouchGui")
        if screenGui then
            local touchFrame = screenGui:FindFirstChild("TouchControlFrame")
            if touchFrame then
                local jumpBtn = touchFrame:FindFirstChild("JumpButton")
                if jumpBtn then
                    _jumpConns[#_jumpConns+1] = jumpBtn.InputBegan:Connect(function()
                        if jumpPowerEnabled then
                            _jumpHolding = true
                            humanoid.Jump = true
                        end
                    end)
                    _jumpConns[#_jumpConns+1] = jumpBtn.InputEnded:Connect(function()
                        _jumpHolding = false
                    end)
                end
            end
        end
    end

    if PlayerGui:FindFirstChild("TouchGui") then
        hookJumpButton()
    else
        _jumpConns[#_jumpConns+1] = PlayerGui.ChildAdded:Connect(function(child)
            if child.Name == "TouchGui" then
                task.wait(0.1)
                hookJumpButton()
            end
        end)
    end

    ContextActionService:BindAction("_HJ_FlyUp", function(_, inputState, inputObj)
        if not jumpPowerEnabled then return Enum.ContextActionResult.Pass end
        if inputObj.UserInputType == Enum.UserInputType.Keyboard
            and inputObj.KeyCode == Enum.KeyCode.Space then
            if inputState == Enum.UserInputState.Begin then
                _jumpHolding = true
                humanoid.Jump = true
                return Enum.ContextActionResult.Sink
            elseif inputState == Enum.UserInputState.End then
                _jumpHolding = false
                return Enum.ContextActionResult.Sink
            end
        end
        return Enum.ContextActionResult.Pass
    end, false, Enum.KeyCode.Space)

    _jumpConns[#_jumpConns+1] = RunService.RenderStepped:Connect(function()
        if jumpPowerEnabled and _jumpHolding and hrp then
            hrp.Velocity = Vector3.new(hrp.Velocity.X, _jumpFlySpeed, hrp.Velocity.Z)
        end
    end)

    _jumpConns[#_jumpConns+1] = humanoid:GetPropertyChangedSignal("Jump"):Connect(function()
        if not humanoid.Jump then end
    end)
end

LocalPlayer.CharacterAdded:Connect(function(char)
    if jumpPowerEnabled then setupHighJump(char) end
end)

        return origTween(obj, info, target)
    end
end

local function hookButton(btn)
    if not btn then return end
    if btn:FindFirstChild("UnlocksAtText") then btn.UnlocksAtText.Visible = false end
    if btn:FindFirstChild("EmoteName")     then btn.EmoteName.Visible = true      end
    CoreUI.on_click(btn, function()
        local hum = CharModule.get_hum()
        if not hum or hum.Health <= 0 then return end
        if EmotesUI.current_emote_playing.get() == btn then
            EmotesUI.current_emote_playing.set(nil)
        else
            EmotesUI.current_emote_playing.set(btn)
        end
        task.wait(0.12)
        EmotesUI.enabled.set(false)
    end)
    EmotesUI.current_emote_playing.hook(function(current)
        if btn:FindFirstChild("EmoteEquipped") then
            btn.EmoteEquipped.Visible = (current == btn)
        end
    end)
end

local AutoHealEnabled   = false
local AutoHealThreshold = 60
local AutoHealCooldown  = 0.05

local _AutoHealLastAttempt = 0
local _healInProgress      = false
local _infoCache           = {}
local _infoCacheTick       = 0

local function AH_GetInfo(itemName)
    if tick() - _infoCacheTick > 0.2 then
        _infoCache    = {}
        _infoCacheTick = tick()
    end
    if _infoCache[itemName] then return _infoCache[itemName] end

    local amount, IsHaving, Uid = 0, false, nil
    local itemsFrame = LocalPlayer.PlayerGui:FindFirstChild("Items")
    if itemsFrame then
        local holder    = itemsFrame:FindFirstChild("ItemsHolder")
        local scrolling = holder and holder:FindFirstChild("ItemsScrollingFrame")
        if scrolling then
            for _, v in pairs(scrolling:GetChildren()) do
                if v.Name ~= "Folder" and v.Name ~= "UIGridLayout" and v.Name ~= "ItemTemplate" then
                    local nameLabel = v:FindFirstChild("ItemName")
                    if nameLabel and nameLabel.Text == itemName then
                        Uid = v.Name
                        amount += 1
                        IsHaving = true
                        break
                    end
                end
            end
        end
    end

    local result = { [1] = amount, [2] = IsHaving, [3] = Uid }
    _infoCache[itemName] = result
    return result
end

local function AH_UseBloodBag()
    if _healInProgress then return false end
    _healInProgress = true

    _infoCache = {}
    _infoCacheTick = 0
    local bloodBagInfo = AH_GetInfo("Blood Bag")
    if not bloodBagInfo[2] then _healInProgress = false return false end

    local bloodBagUid = bloodBagInfo[3]
    local char        = LocalPlayer.Character
    local humanoid    = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not humanoid then _healInProgress = false return false end

    pcall(function() netGet("toggle_equip_item", bloodBagUid) end)

    local bagInChar = nil
    local elapsed   = 0
    while elapsed < 0.4 do
        task.wait(0.02)
        elapsed += 0.02
        bagInChar = char:FindFirstChild("Blood Bag")
            or (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Blood Bag"))
        if bagInChar then break end
    end

    if not bagInChar then
        local bp = LocalPlayer:FindFirstChild("Backpack")
        if bp then
            for _, tool in pairs(bp:GetChildren()) do
                if tool.Name == "Blood Bag" then
                    bagInChar = tool
                    break
                end
            end
        end
    end

    if not bagInChar then _healInProgress = false return false end

    if bagInChar.Parent ~= char then
        pcall(function() humanoid:EquipTool(bagInChar) end)
        task.wait(0.02)
        bagInChar = char:FindFirstChild("Blood Bag")
    end

    if not bagInChar then _healInProgress = false return false end

    local ok = pcall(function() netGet("consume_power_up", bagInChar) end)

    task.wait(0.05)
    pcall(function() humanoid:UnequipTools() end)

    _healInProgress = false
    return ok
end

        local ok, frame = pcall(function()
            return LocalPlayer.PlayerGui.Items.ItemsHolder.ItemsScrollingFrame
        end)
        if not ok or not frame then return nil end
        for _, v in ipairs(frame:GetChildren()) do
            if v.Name ~= "Folder" and v.Name ~= "UIGridLayout" and v.Name ~= "ItemTemplate" then
                local itemType = v:GetAttribute("ItemType")
                if itemType and (itemType == "bike" or itemType == "bmx") then
                    return v.Name
                end
            end
        end
        return nil
    end

        local vf = Workspace:FindFirstChild("Vehicles")

        for _, v in ipairs(vf:GetChildren()) do
            if v:GetAttribute("OwnerUserId") == LocalPlayer.UserId then
                return v
            end
        end
        return nil
    end

    local function SnapWithBike()
        local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        local hrp  = char:WaitForChild("HumanoidRootPart")
        local hum  = char:WaitForChild("Humanoid")

        local vehicle = getMyVehicle()

        if not vehicle then
            local uid = getBikeUid()
            if uid then
                pcall(function() netGet("toggle_equip_item", uid) end)
                task.wait(0.3)
                vehicle = getMyVehicle()
            end
        end

        if not vehicle then
            warn("[Snap Vehicle] ไม่พบรถของคุณ")
            return
        end

        local originalCF = vehicle:GetPrimaryPartCFrame()

        if not hum.Sit then
            local chassis  = vehicle:FindFirstChild("Chassis")
            local dpAttach = chassis and chassis:FindFirstChild("DrivePromptAttachment")
            local dp       = dpAttach and dpAttach:FindFirstChild("DrivePrompt")

            if dp then
                for _ = 1, 2 do
                    if hum.Sit then break end
                    pcall(function() if fireproximityprompt then fireproximityprompt(dp, 1) end end)
                    local t = tick()
                    repeat task.wait(0.03) until hum.Sit or (tick() - t > 0.15)
                end
            end

            if not hum.Sit then
                originalCF = vehicle:GetPrimaryPartCFrame()
                vehicle:SetPrimaryPartCFrame(CFrame.new(hrp.Position + Vector3.new(0, 2, -5)))
                local chassis2  = vehicle:FindFirstChild("Chassis")
                local dpAttach2 = chassis2 and chassis2:FindFirstChild("DrivePromptAttachment")
                local dp2       = dpAttach2 and dpAttach2:FindFirstChild("DrivePrompt")
                if dp2 then
                    for _ = 1, 2 do
                        if hum.Sit then break end
                        pcall(function() if fireproximityprompt then fireproximityprompt(dp2, 1) end end)
                        local t = tick()
                        repeat task.wait(0.03) until hum.Sit or (tick() - t > 0.15)
                    end
                end
            end
        end

        if not hum.Sit then
            warn("[Snap Vehicle] ขึ้นรถไม่สำเร็จ")
            return
        end

        local targetPos = hrp.Position + Vector3.new(0, SNAP_VEH_DEPTH, 0)
        vehicle:SetPrimaryPartCFrame(CFrame.new(targetPos))
        hrp.CFrame = CFrame.new(targetPos + Vector3.new(0, 1, 0))
        hum.PlatformStand = true

        task.wait(0.17)

        pcall(function() netGet("exit_seat") end)
        hum.Sit = false
        hum.PlatformStand = false
        hum:ChangeState(Enum.HumanoidStateType.Jumping)
        vehicle:SetPrimaryPartCFrame(originalCF)
    end

    local function PullMyCar()
        local char = LocalPlayer.Character
        if not char then return end
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end

        local Vehicles = Workspace:FindFirstChild("Vehicles")
        if not Vehicles then
            warn("[Pull Car] ไม่เจอโฟลเดอร์ Vehicles")
            return
        end

        local pulled = false
        for _, vehicle in pairs(Vehicles:GetChildren()) do
            if vehicle:IsA("Model") then
                local ownerId = vehicle:GetAttribute("OwnerUserId")
                if ownerId and ownerId == LocalPlayer.UserId then
                    local offset = CFrame.new(
                        math.random(-4, 4),
                        3,
                        math.random(-12, -6)
                    )
                    if vehicle.PrimaryPart then
                        vehicle:SetPrimaryPartCFrame(rootPart.CFrame * offset)
                    else
                        local part = vehicle:FindFirstChildWhichIsA("BasePart")
                        if part then
                            part.CFrame = rootPart.CFrame * offset
                        end
                    end
                    pulled = true
                end
            end
        end

        if not pulled then
            warn("[Pull Car] ไม่พบรถของตัวเอง")
        end
    end

        for _, v in next, Vehicles:GetChildren() do
            if v:GetAttribute("OwnerUserId") == LocalPlayer.UserId then
                return v
            end
        end
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.SeatPart then return hum.SeatPart.Parent end
        end
        return nil
    end

    local VM = {
        AttributeCache = setmetatable({}, { __mode = "k" }),
        ChassisCache   = setmetatable({}, { __mode = "k" }),
        LastVehicle    = nil,
    }

    local function VM_DefaultSnap(part)
        return { hadCustom=false, density=nil, densityFallback=1, friction=0.3, elasticity=0, frictionWeight=1, elasticityWeight=1 }
    end

    function VM:Valid(inst)
        return inst ~= nil and typeof(inst) == "Instance" and inst.Parent ~= nil
    end

    function VM:StoreOriginal(inst, attr, val)
        if not self:Valid(inst) then return end
        self.AttributeCache[inst] = self.AttributeCache[inst] or {}
        if self.AttributeCache[inst][attr] == nil then
            self.AttributeCache[inst][attr] = val
        end
    end

    function VM:CaptureChassisState(chassis)
        local cur = chassis.CustomPhysicalProperties
        if cur then
            return { hadCustom=true, density=cur.Density, friction=cur.Friction, elasticity=cur.Elasticity, frictionWeight=cur.FrictionWeight, elasticityWeight=cur.ElasticityWeight, densityFallback=cur.Density }
        end
        return VM_DefaultSnap(chassis)
    end

    function VM:RestoreChassisState(chassis, cache)
        if not cache then return end
        if cache.hadCustom and cache.density then
            chassis.CustomPhysicalProperties = PhysicalProperties.new(cache.density, cache.friction, cache.elasticity, cache.frictionWeight, cache.elasticityWeight)
        else
            pcall(function() chassis.CustomPhysicalProperties = nil end)
        end
    end

    function VM:ApplyFolder(folder)
        if not self:Valid(folder) then return end
        for _, info in ipairs(VM_Storage.Vehicle_Attributes) do
            local cur = folder:GetAttribute(info.Name)
            if cur ~= nil then
                self:StoreOriginal(folder, info.Name, cur)
                local new = VM_Option("VehicleMods_" .. info.Name)
                if new then folder:SetAttribute(info.Name, new) end
            end
        end
        local tm = VM_Option("VehicleMods_TorqueMultiplier") or 1
        for _, attr in ipairs({"maxSpeedTorque","minSpeedTorque"}) do
            local cur = folder:GetAttribute(attr)
            if cur ~= nil then
                self:StoreOriginal(folder, attr, cur)
                local orig = self.AttributeCache[folder] and self.AttributeCache[folder][attr]
                if type(orig) == "number" then folder:SetAttribute(attr, orig * tm) end
            end
        end
    end

    function VM:ApplySlam(vehicle)
        local chassis = vehicle:FindFirstChild("Chassis")
        if not (chassis and chassis:IsA("BasePart")) then return end
        if VM_Config.Slam then
            if not self.ChassisCache[chassis] then
                self.ChassisCache[chassis] = self:CaptureChassisState(chassis)
            end
            local c = self.ChassisCache[chassis] or VM_DefaultSnap(chassis)
            chassis.CustomPhysicalProperties = PhysicalProperties.new(50, c.friction or 0.3, c.elasticity or 0, c.frictionWeight or 1, c.elasticityWeight or 1)
        else
            local c = self.ChassisCache[chassis]
            if c then self:RestoreChassisState(chassis, c); self.ChassisCache[chassis] = nil end
        end
    end

    function VM:Revert(vehicle)
        for inst, attrMap in pairs(self.AttributeCache) do
            local should = (not vehicle) or (self:Valid(inst) and inst:IsDescendantOf(vehicle))
            if should then
                if self:Valid(inst) then
                    for attr, orig in pairs(attrMap) do pcall(inst.SetAttribute, inst, attr, orig) end
                end
                self.AttributeCache[inst] = nil
            end
        end
        for chassis, cache in pairs(self.ChassisCache) do
            local should = (not vehicle) or (self:Valid(chassis) and chassis:IsDescendantOf(vehicle))
            if should then
                if self:Valid(chassis) then self:RestoreChassisState(chassis, cache) end
                self.ChassisCache[chassis] = nil
            end
        end
    end

    function VM:Step()
        if not VM_Config.Enabled then
            if self.LastVehicle then self:Revert(self.LastVehicle); self.LastVehicle = nil end
            self:Revert(nil)
            return
        end
        local vehicle = VM_FindVehicle()
        if not (vehicle and vehicle.Parent) then
            if self.LastVehicle then self:Revert(self.LastVehicle); self.LastVehicle = nil end
            return
        end
        if self.LastVehicle and self.LastVehicle ~= vehicle then self:Revert(self.LastVehicle) end
        self.LastVehicle = vehicle
        for _, child in ipairs(vehicle:GetChildren()) do
            if child:IsA("Folder") then self:ApplyFolder(child) end
        end
        self:ApplySlam(vehicle)
    end

    task.spawn(function()
        while task.wait(0.25) do VM:Step() end
    end)

    task.spawn(function()
        while task.wait(0.5) do
            if not VM_Config.Enabled then continue end
            local vehicle = VM_FindVehicle()
            if not vehicle then continue end
            for _, child in next, vehicle:GetChildren() do
                if child:IsA("Folder") then
                    for attr in next, child:GetAttributes() do
                        if attr == "forwardMaxSpeed" then
                            if not child:GetAttribute("OldforwardMaxSpeed") then
                                child:SetAttribute("OldforwardMaxSpeed", child:GetAttribute("forwardMaxSpeed"))
                            end
                            child:SetAttribute("forwardMaxSpeed", child:GetAttribute("OldforwardMaxSpeed") * VM_Config.MaxSpeedMultiplier)
                        end
                    end
                end
            end
        end
    end)

    local VehicleModToggle = VehicleTab:Toggle({
        Title    = "Vehicle Mods",
        Desc     = "เพิ่ม Acceleration, Torque, Max Speed",
        Default  = false,
        Callback = function(v)
            VM_Config.Enabled = v
            if not v then VM:Revert(nil); VM.LastVehicle = nil end
        end,
    })

    VehicleTab:Toggle({
        Title    = "Slam",
        Desc     = "ทำให้รถต่ำลง (ต้องเปิด Vehicle Mods)",
        Default  = false,
        Callback = function(v)
            VM_Config.Slam = v
            if not v then
                for chassis, cache in pairs(VM.ChassisCache) do
                    if VM:Valid(chassis) then VM:RestoreChassisState(chassis, cache) end
                    VM.ChassisCache[chassis] = nil
                end
            end
        end,
    })

    VehicleTab:Slider({
        Title = "Acceleration",
        Step  = 1,
        Value = { Min = 5, Max = 100, Default = 30 },
        Callback = function(v) VM_Config.Acceleration = v end,
    })

    VehicleTab:Slider({
        Title = "Slip Threshold",
        Step  = 1,
        Value = { Min = 10, Max = 100, Default = 50 },
        Callback = function(v) VM_Config.SlipThreshold = v end,
    })

    VehicleTab:Slider({
        Title = "Static Friction",
        Step  = 1,
        Value = { Min = 1, Max = 10, Default = 5 },
        Callback = function(v) VM_Config.StaticFriction = v end,
    })

    VehicleTab:Slider({
        Title = "Torque Multiplier",
        Step  = 0.1,
        Value = { Min = 0, Max = 5, Default = 2 },
        Callback = function(v) VM_Config.TorqueMultiplier = v end,
    })

    VehicleTab:Slider({
        Title = "Max Speed Multiplier",
        Step  = 0.05,
        Value = { Min = 1, Max = 1.3, Default = 1.2 },
        Callback = function(v) VM_Config.MaxSpeedMultiplier = v end,
    })
end
